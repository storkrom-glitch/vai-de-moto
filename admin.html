function renderMoto(){ const el = document.getElementById('aba-moto'); if(!el) return; const lista = listaMoto().filter(m => (m.nome||'').toLowerCase().includes(buscaM.toLowerCase()) || (m.telefone||'').includes(buscaM) || (m.placa||'').toLowerCase().includes(buscaM.toLowerCase()) ).sort((a,b)=>(a.nome||'').localeCompare(b.nome||'')); const stats = id => { const cs = listaCorridas().filter(c=>c.motoqueiroId===id && c.status==='concluida'); return { n: cs.length, tot: cs.reduce((t,c)=>t+(parseFloat(c.valorAceito)||0),0) }; }; el.innerHTML = `
${lista.map(m=>{ const s = stats(m.id); return `
🏍️ ${esc(m.nome||'—')} ${m.bloqueado?'BLOQUEADO':'ATIVO'}
📞 ${esc(m.telefone||'—')} • 🪪 CNH ${esc(m.cnh||'—')}
🛵 ${esc(m.modelo||'—')} ${esc(m.cor||'')} • placa ${esc(m.placa||'—')} (${esc(m.ano||'—')})
💠 PIX: ${esc(m.pix||'não cadastrada')} • desde ${dataFmt(m.criadoEm)}
✅ ${s.n} corrida(s) • 💰 ${brl(s.tot)}
${m.bloqueado ? `` : ``}
`; }).join('') || '
Nenhum motoqueiro encontrado.
'} `; } window.limparBuscaMoto = () => { buscaM = ''; const inp = document.getElementById('buscaMotoInput'); if(inp) inp.value = ''; renderMoto(); };
