function renderPass(){ const el = document.getElementById('aba-pass'); if(!el) return; const lista = listaPass().filter(p => (p.nome||'').toLowerCase().includes(buscaP.toLowerCase()) || (p.telefone||'').includes(buscaP) ).sort((a,b)=>(a.nome||'').localeCompare(b.nome||'')); const corridasDe = id => listaCorridas().filter(c=>c.passageiroId===id && c.status==='concluida').length; el.innerHTML = `
${lista.map(p=>`
🙋 ${esc(p.nome||'—')} ${p.bloqueado?'BLOQUEADO':'ATIVO'}
📞 ${esc(p.telefone||'—')} • desde ${dataFmt(p.criadoEm)} • ${corridasDe(p.id)} corrida(s) concluída(s)
${p.bloqueado ? `` : ``}
`).join('') || '
Nenhum passageiro encontrado.
'} `; } window.limparBuscaPass = () => { buscaP = ''; const inp = document.getElementById('buscaPassInput'); if(inp) inp.value = ''; renderPass(); };
